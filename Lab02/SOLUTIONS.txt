class BankAccount(object): # this is the class header
    def __init__(self, name, balance=0):
        self.__name = name
        self.__balance = balance
        self.__withdraw_override = False

    def set_name(self, name):
        self.__name = name

    def get_name(self):
        return self.__name

    def enable_override(self):
        self.__withdraw_override = True

    def disable_override(self):
        self.__withdraw_override = False

    def deposit(self, amount):
        if isinstance(amount, int) or isinstance(amount, float):
            self.__balance += amount
        else:
            print("Balance unchanged.")

    def withdraw(self, amount):
        if isinstance(amount, int) or isinstance(amount, float):
            if not self.__withdraw_override and amount > self.__balance:
                print("Balance exceeded. Withdrawing available funds.")
                amount = self.__balance
            self.__balance -= amount
        else:
            print("Balance unchanged.")
        return amount

    # INSERT THE SIX DUNDER METHODS BELOW THIS COMMENT LINE

    def __lt__(self, other): # returns True if this account's name is less than the other account's name - otherwise, return None (signalling failure)
        if isinstance(other, BankAccount):
            return self.__name < other.get_name()

    def __eq__(self, other):
        if isinstance(other, BankAccount):
            return self.__name == other.get_name()

    def __gt__(self, other):
        if isinstance(other, BankAccount):
            return self.__name > other.get_name()

    def __le__(self, other):
        if isinstance(other, BankAccount):
            return self.__name <= other.get_name()

    def __ge__(self, other):
        if isinstance(other, BankAccount):
            return self.__name >= other.get_name()

    def __ne__(self, other):
        if isinstance(other, BankAccount):
            return self.__name != other.get_name()

    # INSERT THE SIX DUNDER METHODS ABOVE THIS COMMENT LINE

    def __str__(self):
        return_value = ""
        return_value += f"Name: {self.__name}\n"
        return_value += f"Balance: {self.__balance}"
        return return_value

# WRITE YOUR BankAccounts CLASS BELOW THIS LINE

class BankAccounts(object):
    def __init__(self):
        self.__account_list = []

    def add_account(self, account):
        if isinstance(account, BankAccount):
            self.__account_list.append(account)

    def remove_account(self, account):
        if isinstance(account, BankAccount):
            try:
                self.__account_list.remove(account)
            except ValueError:
                return False
            return True

    def bank_total(self):
        total = 0
        for account in self.__account_list:
            total += account.get_balance()
        return total

    def __str__(self):
        return_value = ""
        for account in self.__account_list:
            return_value += f"{account}\n"
        return return_value
